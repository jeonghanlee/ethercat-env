# cyclictest - primary scheduling-latency probe. Run on an RT kernel, after the
# RT limits and isolated CPUs are in place, with the EtherCAT master idle.
cyclictest --mlockall --smp --priority=80 --interval=200 --distance=0

# hwlatdetect - hardware/SMI-induced latency probe. Detects firmware stalls the
# scheduler cannot see.
hwlatdetect --duration=60 --threshold=10
